  <action>
    Create `apps/web/components/home/GenreChips.tsx` as a server component (no "use client"):

    ```tsx
    import Link from "next/link";
    import type { GenreChipsData } from "./types";

    interface Props {
      section: GenreChipsData;
    }

    export function GenreChips({ section }: Props) {
      if (section.status !== "ready" || section.items.length === 0) return null;

      return (
        <nav aria-label="Browse by genre" className="...">
          <div className="flex gap-2.5 overflow-x-auto pb-1 scrollbar-hide ...">
            {section.items.map((chip) => (
              <Link
                key={chip.id}
                href={chip.href}
                prefetch={false}
                className="..."
              >
                <span ...>{chip.emoji}</span>
                <span ...>{chip.label}</span>
              </Link>
            ))}
          </div>
        </nav>
      );
    }
    ```

    Design spec:
    - `<nav>` wrapper with appropriate aria-label
    - Inner `<div>` — `flex gap-2.5 overflow-x-auto` — single horizontal scrolling row; hide scrollbar with `[&::-webkit-scrollbar]:hidden [-ms-overflow-style:none] [scrollbar-width:none]` (Tailwind arbitrary variants)
    - Each chip: `inline-flex flex-shrink-0 items-center gap-1.5 rounded-full border border-white/10 bg-white/5 px-4 py-2 text-[13px] font-semibold text-white/80 backdrop-blur-sm transition-colors hover:border-primary/40 hover:bg-primary/10 hover:text-primary`
    - Emoji span: `text-[16px] leading-none` (prevents emoji from being clipped)
    - Label span: `whitespace-nowrap`
    - No `flex-wrap` — chips MUST stay on one line and scroll
  </action>
