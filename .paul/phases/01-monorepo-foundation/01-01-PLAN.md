<tasks>

<task type="auto">
  <name>Task 1: Create pnpm-workspace.yaml and turbo.json</name>
  <files>pnpm-workspace.yaml, turbo.json</files>
  <action>
    Create `pnpm-workspace.yaml` at repo root:
    ```yaml
    packages:
      - "apps/*"
      - "packages/*"
    ```

    Create `turbo.json` at repo root:
    ```json
    {
      "$schema": "https://turbo.build/schema.json",
      "tasks": {
        "build": {
          "dependsOn": ["^build"],
          "outputs": [".next/**", "!.next/cache/**", "dist/**"]
        },
        "dev": {
          "persistent": true,
          "cache": false
        },
        "typecheck": {
          "dependsOn": ["^build"]
        },
        "lint": {
          "outputs": []
        }
      }
    }
    ```

    `^build` means "build my workspace dependencies first".
    `persistent: true` on dev keeps the process alive (Next.js dev server, Metro bundler).
    Do NOT add a `globalDependencies` field — not needed at this stage.
  </action>
  <verify>cat pnpm-workspace.yaml && cat turbo.json — both files present and valid JSON/YAML</verify>
  <done>AC-1 and AC-2 satisfied: workspace declared, pipeline defined</done>
</task>
